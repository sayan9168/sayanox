// compiler_min.sa - pure Sayanox mini-compiler (generation 1 source).
//
// The C seed (selfhost/seed/sxc_seed.c) compiles this file into selfhost/gen1.c.
// gen1 accepts the "pure min" target language and emits standalone C:
//
//   hold NAME = NUMBER | NAME | "text" | NAME + NUMBER | call(...)
//   show NUMBER | NAME | "text" | call(...)
//   while NAME OP NUMBER|NAME { ... }
//   when  NAME OP NUMBER|NAME { ... } otherwise { ... }
//   // line comments
//
//   OP is one of  ==  !=  <  <=  >  >=
//
// String builtins exposed to target programs (only what target programs need):
//   concat / len / sx_index / chr / read_file / write_file / arg / arg_count
// The runtime-name aliases (sx_cat, sx_len, sx_idx, sx_chr, sx_read, sx_write,
// sx_arg, sx_argc) are accepted too so that a child compiler can copy call
// text without a name table.
//
// Design notes
//   * The whole compiler is ONE flat top-level program.  The seed has no
//     closures, so a `make` helper could not see pos/body: there are none.
//   * Statements are recognised by a single character scanner with a brace
//     depth counter.  Bodies are NOT skipped by a nested parser, so hold and
//     show inside while/when bodies are compiled as well.
//   * Numeric programs get clean standalone C (`double n = 0;`, `n = n + 1;`,
//     `while (n < 3) {`).  Programs that use string calls get a small static
//     helper block emitted above main; no external runtime header is needed.
//   * First `hold name = ...` declares (`double` or `char *`), later ones
//     assign.  Membership is tracked in the "|name|" strings decls/sdecls.
//
// See docs/STATUS.md for the exact scope, and selfhost/compiler_boot.sa for a
// reduced pure-min compiler written in the subset this compiler accepts.

// ---------------- command line ----------------
hold in_path = "selfhost/mini_in.sa"
hold out_path = "selfhost/mini_out.c"
hold na = arg_count()
when na >= 2 {
  hold in_path = arg(1)
}
when na >= 3 {
  hold out_path = arg(2)
}

// ---------------- compiler state ----------------
hold source = read_file(in_path)
hold n = len(source)
hold pos = 0
hold body = ""
hold decls = "|"
hold sdecls = "|"
hold depth = 0
hold running = 1
hold holds = 0
hold shows = 0
hold whiles = 0
hold whens = 0
hold rt = 0

// ---------------- scratch ----------------
hold key = ""
hold nm = ""
hold nm2 = ""
hold op = ""
hold operand = ""
hold opnd = ""
hold oq = 0
hold nm3 = ""
hold rhs = ""
hold opch = 0
hold argv_txt = ""
hold numtxt = ""
hold ind = ""
hold w9 = ""
hold out = ""
hold q = chr(34)
hold nl = chr(10)
hold bs = chr(92)
hold act = 0
hold c = 0
hold c1 = 0
hold c2 = 0
hold c3 = 0
hold cs = 0
hold cq = 0
hold cn = 0
hold cd = 0
hold cw = 0
hold sk = 0
hold rn = 0
hold skipc = 0
hold skipw = 0
hold wsc = 0
hold isspace = 0
hold inq = 0
hold iscall = 0
hold iswhile = 0
hold rk = 0
hold fk = 0
hold wi = 0
hold wk = 0
hold wfound = 0
hold d2 = 0
hold found = 0
hold sfind = 0
hold di = 0
hold dl = 0
hold kl = 0
hold j = 0
hold ok = 0
hold aa = 0
hold bb = 0
hold pdepth = 0
hold argv_c = ""
hold ai = 0
hold al = 0
hold arun = 0
hold ac = 0
hold islet = 0
hold atok = ""
hold tokrun = 0

// =====================================================================
// main scan loop
// =====================================================================
while running == 1 {

  // ---- phase 1: skip blanks and // comments ----
  hold skipw = 1
  while skipw == 1 {
    when pos >= n {
      hold skipw = 0
      hold running = 0
    } otherwise {
      hold c = sx_index(source,pos)
      hold isspace = 0
      when c == 32 || c == 9 || c == 13 || c == 10 {
        hold isspace = 1
      }
      when isspace == 1 {
        hold pos = pos + 1
      } otherwise {
        when c == 47 {
          hold c2 = 0
          when pos + 1 < n {
            hold c2 = sx_index(source,pos + 1)
          }
          when c2 == 47 {
            // line comment: run to the end of the line
            hold pos = pos + 2
            hold skipc = 1
            while skipc == 1 {
              when pos >= n {
                hold skipc = 0
                hold running = 0
              } otherwise {
                hold c3 = sx_index(source,pos)
                when c3 == 10 {
                  hold skipc = 0
                } otherwise {
                  hold pos = pos + 1
                }
              }
            }
          } otherwise {
            hold pos = pos + 1
          }
        } otherwise {
          hold skipw = 0
        }
      }
    }
  }

  // ---- phase 2: one statement ----
  when running == 1 {
    hold c = sx_index(source,pos)

    // dispatch on the first character; act is 0 for "not a statement"
    hold act = 0
    when c == 104 {
      hold act = 1
    }
    when c == 115 {
      hold act = 2
    }
    when c == 119 {
      hold act = 3
    }
    when c == 125 {
      hold act = 4
    }

    // ================= hold NAME = RHS =================
    when act == 1 {
      hold pos = pos + 4
      // blank skip
      hold sk = 1
      while sk == 1 {
        when pos >= n {
          hold sk = 0
        } otherwise {
          hold cs = sx_index(source,pos)
          when cs == 32 || cs == 9 || cs == 13 {
            hold pos = pos + 1
          } otherwise {
            hold sk = 0
          }
        }
      }
      // name
      hold nm = ""
      hold rn = 1
      while rn == 1 {
        when pos >= n {
          hold rn = 0
        } otherwise {
          hold cn = sx_index(source,pos)
          when (cn >= 97 && cn <= 122) || (cn >= 65 && cn <= 90) || (cn >= 48 && cn <= 57) || cn == 95 {
            hold nm = concat(nm, chr(cn))
            hold pos = pos + 1
          } otherwise {
            hold rn = 0
          }
        }
      }
      // blanks then '='
      hold sk = 1
      while sk == 1 {
        when pos >= n {
          hold sk = 0
        } otherwise {
          hold cs = sx_index(source,pos)
          when cs == 32 || cs == 9 || cs == 13 {
            hold pos = pos + 1
          } otherwise {
            when cs == 61 {
              hold pos = pos + 1
            }
            hold sk = 0
          }
        }
      }
      // blanks
      hold sk = 1
      while sk == 1 {
        when pos >= n {
          hold sk = 0
        } otherwise {
          hold cs = sx_index(source,pos)
          when cs == 32 || cs == 9 || cs == 13 {
            hold pos = pos + 1
          } otherwise {
            hold sk = 0
          }
        }
      }

      // ---------- right hand side ----------
      hold rhs = ""
      hold rk = 0
      hold c1 = 0
      when pos < n {
        hold c1 = sx_index(source,pos)
      }

      // RHS: string literal
      when c1 == 34 {
        hold pos = pos + 1
        hold rhs = q
        hold rn = 1
        while rn == 1 {
          when pos >= n {
            hold rn = 0
          } otherwise {
            hold cq = sx_index(source,pos)
            when cq == 34 {
              hold pos = pos + 1
              hold rhs = concat(rhs, q)
              hold rn = 0
            } otherwise {
              when cq == 92 {
                hold rhs = concat(rhs, bs)
                hold pos = pos + 1
              } otherwise {
                hold rhs = concat(rhs, chr(cq))
                hold pos = pos + 1
              }
            }
          }
        }
        hold rk = 1
        hold rt = 1
      }

      // RHS: number
      when (c1 >= 48 && c1 <= 57) || c1 == 45 {
        hold numtxt = ""
        when c1 == 45 {
          hold numtxt = "-"
          hold pos = pos + 1
        }
        hold rn = 1
        while rn == 1 {
          when pos >= n {
            hold rn = 0
          } otherwise {
            hold cd = sx_index(source,pos)
            when cd >= 48 && cd <= 57 {
              hold numtxt = concat(numtxt, chr(cd))
              hold pos = pos + 1
            } otherwise {
              hold rn = 0
            }
          }
        }
        hold rhs = numtxt
        hold rk = 0
      }

      // RHS: name, name + number, or call(...)
      when (c1 >= 97 && c1 <= 122) || (c1 >= 65 && c1 <= 90) {
        hold nm2 = ""
        hold rn = 1
        while rn == 1 {
          when pos >= n {
            hold rn = 0
          } otherwise {
            hold cn = sx_index(source,pos)
            when (cn >= 97 && cn <= 122) || (cn >= 65 && cn <= 90) || (cn >= 48 && cn <= 57) || cn == 95 {
              hold nm2 = concat(nm2, chr(cn))
              hold pos = pos + 1
            } otherwise {
              hold rn = 0
            }
          }
        }
        hold iscall = 0
        hold c2 = 0
        when pos < n {
          hold c2 = sx_index(source,pos)
        }
        when c2 == 40 {
          hold iscall = 1
        }

        // ---- call(...) ----
        when iscall == 1 {
          hold pos = pos + 1
          hold argv_txt = ""
          hold inq = 0
          hold pdepth = 0
          hold rn = 1
          while rn == 1 {
            when pos >= n {
              hold rn = 0
            } otherwise {
              hold cn = sx_index(source,pos)
              when cn == 41 && inq == 0 && pdepth == 0 {
                hold rn = 0
              } otherwise {
                hold pos = pos + 1
                when inq == 1 {
                  hold argv_txt = concat(argv_txt, chr(cn))
                  when cn == 34 {
                    hold inq = 0
                  }
                  when cn == 92 {
                    when pos < n {
                      hold cq = sx_index(source,pos)
                      hold argv_txt = concat(argv_txt, chr(cq))
                      hold pos = pos + 1
                    }
                  }
                } otherwise {
                  when cn == 40 {
                    hold pdepth = pdepth + 1
                  }
                  when cn == 41 {
                    hold pdepth = pdepth - 1
                  }
                  when cn == 34 {
                    hold inq = 1
                    hold argv_txt = concat(argv_txt, chr(cn))
                  } otherwise {
                    when cn == 32 || cn == 9 {
                    } otherwise {
                      hold argv_txt = concat(argv_txt, chr(cn))
                    }
                  }
                }
              }
            }
          }
          hold pos = pos + 1
          // which builtin?
          // Normalize builtin identifiers inside argument text for nested calls.
          hold argv_c = ""
          hold ai = 0
          hold al = len(argv_txt)
          hold arun = 1
          while arun == 1 {
            when ai >= al {
              hold arun = 0
            } otherwise {
              hold ac = sx_index(argv_txt,ai)
              hold islet = 0
              when (ac >= 97 && ac <= 122) || (ac >= 65 && ac <= 90) || ac == 95 {
                hold islet = 1
              }
              when islet == 1 {
                hold atok = ""
                hold tokrun = 1
                while tokrun == 1 {
                  when ai >= al {
                    hold tokrun = 0
                  } otherwise {
                    hold ac = sx_index(argv_txt,ai)
                    when (ac >= 97 && ac <= 122) || (ac >= 65 && ac <= 90) || (ac >= 48 && ac <= 57) || ac == 95 {
                      hold atok = concat(atok, chr(ac))
                      hold ai = ai + 1
                    } otherwise {
                      hold tokrun = 0
                    }
                  }
                }
                when string_eq(atok, "concat") {
                  hold argv_c = concat(argv_c, "sx_cat")
                } otherwise {
                  when string_eq(atok, "len") {
                    hold argv_c = concat(argv_c, "sx_len")
                  } otherwise {
                    when string_eq(atok, "sx_index") {
                      hold argv_c = concat(argv_c, "sx_idx")
                    } otherwise {
                      when string_eq(atok, "chr") {
                        hold argv_c = concat(argv_c, "sx_chr")
                      } otherwise {
                        when string_eq(atok, "read_file") {
                          hold argv_c = concat(argv_c, "sx_read")
                        } otherwise {
                          when string_eq(atok, "write_file") {
                            hold argv_c = concat(argv_c, "sx_write")
                          } otherwise {
                            when string_eq(atok, "arg") {
                              hold argv_c = concat(argv_c, "sx_arg")
                            } otherwise {
                              when string_eq(atok, "arg_count") {
                                hold argv_c = concat(argv_c, "sx_argc")
                              } otherwise {
                                hold argv_c = concat(argv_c, atok)
                              }
                            }
                          }
                        }
                      }
                    }
                  }
                }
              } otherwise {
                hold argv_c = concat(argv_c, chr(ac))
                hold ai = ai + 1
              }
            }
          }
          hold fk = 0
          when string_eq(nm2, "concat") {
            hold fk = 1
          }
          when string_eq(nm2, "len") {
            hold fk = 2
          }
          when string_eq(nm2, "sx_index") {
            hold fk = 3
          }
          when string_eq(nm2, "chr") {
            hold fk = 4
          }
          when string_eq(nm2, "read_file") {
            hold fk = 5
          }
          when string_eq(nm2, "write_file") {
            hold fk = 6
          }
          when string_eq(nm2, "arg") {
            hold fk = 7
          }
          when string_eq(nm2, "arg_count") {
            hold fk = 8
          }
          when string_eq(nm2, "string_eq") {
            hold fk = 9
          }
          when string_eq(nm2, "sx_cat") {
            hold fk = 1
          }
          when string_eq(nm2, "sx_len") {
            hold fk = 2
          }
          when string_eq(nm2, "sx_idx") {
            hold fk = 3
          }
          when string_eq(nm2, "sx_chr") {
            hold fk = 4
          }
          when string_eq(nm2, "sx_read") {
            hold fk = 5
          }
          when string_eq(nm2, "sx_write") {
            hold fk = 6
          }
          when string_eq(nm2, "sx_arg") {
            hold fk = 7
          }
          when string_eq(nm2, "sx_argc") {
            hold fk = 8
          }
          hold rk = 0
          hold rhs = "0"
          when fk == 1 {
            hold rhs = concat("sx_cat(", argv_c)
            hold rhs = concat(rhs, ")")
            hold rk = 1
            hold rt = 1
          }
          when fk == 2 {
            hold rhs = concat("sx_len(", argv_c)
            hold rhs = concat(rhs, ")")
            hold rt = 1
          }
          when fk == 3 {
            hold rhs = concat("sx_idx(", argv_c)
            hold rhs = concat(rhs, ")")
            hold rt = 1
          }
          when fk == 4 {
            hold rhs = concat("sx_chr(", argv_c)
            hold rhs = concat(rhs, ")")
            hold rk = 1
            hold rt = 1
          }
          when fk == 5 {
            hold rhs = concat("sx_read(", argv_c)
            hold rhs = concat(rhs, ")")
            hold rk = 1
            hold rt = 1
          }
          when fk == 6 {
            hold rhs = concat("sx_write(", argv_c)
            hold rhs = concat(rhs, ")")
            hold rt = 1
          }
          when fk == 7 {
            hold rhs = concat("sx_arg(", argv_c)
            hold rhs = concat(rhs, ")")
            hold rk = 1
            hold rt = 1
          }
          when fk == 8 {
            hold rhs = "sx_argc()"
            hold rt = 1
          }
          when fk == 9 {
            hold rhs = concat("sx_eq(", argv_c)
            hold rhs = concat(rhs, ")")
            hold rt = 1
          }
        }

        // ---- plain name, or name + number ----
        when iscall == 0 {
          // is nm2 a string variable?
          hold key = concat("|", nm2)
          hold key = concat(key, "|")
          hold sfind = 0
          hold di = 0
          hold dl = len(sdecls)
          hold kl = len(key)
          while di + kl <= dl {
            hold ok = 1
            hold j = 0
            while j < kl {
              hold aa = sx_index(sdecls, di + j)
              hold bb = sx_index(key, j)
              when aa == bb {
                hold j = j + 1
              } otherwise {
                hold ok = 0
                hold j = kl
              }
            }
            when ok == 1 {
              hold sfind = 1
              hold di = dl
            } otherwise {
              hold di = di + 1
            }
          }
          hold rk = 0
          when sfind == 1 {
            hold rk = 1
          }
          hold rhs = nm2
          // optional "+"/"-" operand: NUMBER or NAME (kept flat, no parser)
          hold sk = 1
          while sk == 1 {
            when pos >= n {
              hold sk = 0
            } otherwise {
              hold cs = sx_index(source,pos)
              when cs == 32 || cs == 9 || cs == 13 {
                hold pos = pos + 1
              } otherwise {
                hold sk = 0
              }
            }
          }
          hold c2 = 0
          when pos < n {
            hold c2 = sx_index(source,pos)
          }
          when c2 == 43 || c2 == 45 {
            hold opch = c2
            hold pos = pos + 1
            hold sk = 1
            while sk == 1 {
              when pos >= n {
                hold sk = 0
              } otherwise {
                hold cs = sx_index(source,pos)
                when cs == 32 || cs == 9 || cs == 13 {
